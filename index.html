Quick Start
Up in 30 seconds
🐍 Python 🐳 Docker 🏳️ Windows 
# Clone & run — no login needed
git clone https://github.com/anshul2414/cyberscan-pro-v2.git
cd cyberscan-pro-v2
pip install -r requirements.txt
python app.py
# Open http://localhost:5000 — no login required!
# Docker Compose
git clone https://github.com/anshul2414/cyberscan-pro-v2.git
cd cyberscan-pro-v2 && docker-compose up -d
# Open http://localhost:8080
# Windows one-click
git clone https://github.com/anshul2414/cyberscan-pro-v2.git
cd cyberscan-pro-v2 && run.bat
# Opens http://localhost:5000
